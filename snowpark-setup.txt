## DDL for Config_table, DQ_ERROR_LOG & QUARANTINE_DQ

CREATE OR REPLACE TABLE CONFIGURATION_MASTER (
    configuration_id   NUMBER AUTOINCREMENT START 1 INCREMENT 1 NOT NULL,
    source_system      STRING,                
    source_object_name STRING,                
    bronze_table       STRING,                
    silver_table       STRING,                
    unified_table      STRING,                
    frequency          STRING,                
    landing_path       STRING,                
    archive_path       STRING,                
    file_format        STRING,                
    delimiter          STRING DEFAULT ',',    
    header             BOOLEAN DEFAULT TRUE,  
    primary_key        STRING,                
    active_flag        BOOLEAN DEFAULT TRUE,  
    created_ts         TIMESTAMP_LTZ DEFAULT CURRENT_TIMESTAMP(),
    updated_ts         TIMESTAMP_LTZ,
    CONSTRAINT PK_CONFIGURATION_MASTER PRIMARY KEY (configuration_id)
);

CREATE OR REPLACE TABLE SNOWFLAKE_LEARNING_DB.PUBLIC.DQ_ERROR_LOG (
    RUN_ID VARCHAR(16777216),
    RUN_TIMESTAMP TIMESTAMP_NTZ(9),
    LAYER VARCHAR(16777216),
    SOURCE_OBJECT VARCHAR(16777216),
    TABLE_NAME VARCHAR(16777216),
    CHECK_TYPE VARCHAR(16777216),
    CHECK_COLUMN VARCHAR(16777216),
    TOTAL_RECORDS NUMBER(38,0),
    FAILED_RECORDS NUMBER(38,0),
    PASS_PERCENTAGE NUMBER(5,2),
    STATUS VARCHAR(16777216),
    MESSAGE VARCHAR(16777216)
);


====================================================================================================

CREATE OR REPLACE TABLE SNOWFLAKE_LEARNING_DB.PUBLIC.QUARANTINE_DQ (
    -- Metadata columns
    QUARANTINE_ID NUMBER AUTOINCREMENT START 1 INCREMENT 1 COMMENT 'Unique identifier for quarantined record',
    QUARANTINE_TIMESTAMP TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP() COMMENT 'Timestamp when record was quarantined',
    SOURCE_OBJECT VARCHAR(255) COMMENT 'Source object/file name',
    DQ_FAILURE_REASON VARCHAR(1000) COMMENT 'Comma-separated list of DQ validation failures',
    -- Audit columns
    CREATED_BY VARCHAR(100) DEFAULT CURRENT_USER() COMMENT 'User who created the record',
    CREATED_DATE TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP() COMMENT 'Record creation timestamp'
)
COMMENT = 'Quarantine table for records failing data quality validation checks';

    TRUE,                              -- header
    'case_id',                         -- primary_key (from source file)
